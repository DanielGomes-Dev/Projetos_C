TARVB *Busca(TARVB* a, int ch){
  if(!a) return NULL;
  int i = 0;
  while(i < a->nchaves && ch > a->chave[i]) i++;
  if(i < a->nchaves && ch == a->chave[i]) return a;
  if(a->folha) return NULL;
  return Busca(a->filho[i], ch);
}

TARVB *Insere(TARVB *T, int k, int t){
  if(Busca(T,k)) return T;
  if(!T){
    T=Cria(t);
    T->chave[0] = k;
    T->nchaves=1;
    return T;
  }
  if(T->nchaves == (2*t)-1){
    TARVB *S = Cria(t); 
    S->nchaves=0;
    S->folha = 0;
    S->filho[0] = T;
    S = Divisao(S,1,T,t);
    S = Insere_Nao_Completo(S,k,t);
    return S;
  }
  T = Insere_Nao_Completo(T,k,t);
  return T;
}

TARVB *Divisao(TARVB *x, int ind, TARVB* y, int t){
  TARVB *z = Cria(t);
  z->nchaves = t-1;
  z->folha = y->folha;
  int j;
  for(j=0;j<t-1;j++) z->chave[j] = y->chave[j+t];
  if(!y->folha){
    for(j=0;j<t;j++){
      z->filho[j] = y->filho[j+t];
      y->filho[j+t] = NULL;
    }

  x->chave[ind-1] = y->chave[t-1];
  x->nchaves++;
  return x;
}

TARVB *Insere_Nao_Completo(TARVB *x, int k, int t){
  int i = x->nchaves-1;
  if(x->folha){
    while((i>=0) && (k<x->chave[i])){
      x->chave[i+1] = x->chave[i];
      i--;

    x->nchaves++;

### Explicação Passo a Passo

O problema resolvido aqui é a manutenção e manipulação de uma **Árvore B** através de um **algoritmo preemptivo** (que corrige a árvore antes que um nó transborde). A regra central da Árvore B é que todo nó (exceto a raiz) deve ter entre **t-1** e **2t-1** chaves.

#### 1. Função `Busca`

* **Passo 1:** `if(!a) return NULL;` – Verifica a validade do nó atual. Se o ponteiro for nulo, a busca falhou.
* **Passo 2:** `while(i < a->nchaves && ch > a->chave[i]) i++;` – Como as chaves estão em ordem crescente, o loop avança o índice `i` até encontrar uma chave maior ou igual à buscada. Isso determina qual subárvore devemos seguir.
* **Passo 3:** `if(i < a->nchaves && ch == a->chave[i]) return a;` – Se a chave procurada foi encontrada exatamente na posição `i`, retorna-se o nó.
* **Passo 4:** `if(a->folha) return NULL;` – Se não achou a chave e o nó é uma folha (não tem filhos), significa que o elemento não existe na árvore.
* **Passo 5:** `return Busca(a->filho[i], ch);` – Chama a função recursivamente descendo para o filho correto.

#### 2. Função `Insere`

* **Passo 1:** `if(Busca(T,k)) return T;` – Impede a inserção de chaves duplicadas.
* **Passo 2:** Criação da raiz `if(!T) { ... }`. Se a árvore está vazia, aloca-se o primeiro nó e insere-se a chave `k`.
* **Passo 3 (O Algoritmo Preemptivo):** `if(T->nchaves == (2*t)-1)` – *"Qual é o problema? Nó no caminho de inserção contendo 2t – 1 chaves."* Se a raiz está completamente cheia, criamos uma nova raiz `S`, atrelamos a raiz antiga `T` como seu filho e chamamos a função `Divisao` para partir `T` ao meio. Só assim a árvore cresce em altura.
* **Passo 4:** Estando a raiz com espaço garantido, chama-se `Insere_Nao_Completo(T,k,t)` ou `Insere_Nao_Completo(S,k,t)`.

#### 3. Função `Divisao`

Esta função divide um nó `y` cheio (com **2t-1** chaves) em dois. A chave do meio sobe para o pai `x`.

* **Passo 1:** Cria-se o nó `z` para receber a metade direita de `y`.
* **Passo 2:** Move-se as últimas **t-1** chaves de `y` para `z` (`for(j=0;j<t-1;j++) z->chave[j] = y->chave[j+t];`).
* **Passo 3:** Se `y` não for folha, os ponteiros dos filhos correspondentes também são transferidos para `z`.
* **Passo 4:** O nó `y` tem seu tamanho reduzido para **t-1**.
* **Passo 5:** Abre-se espaço no nó pai `x` empurrando suas chaves e filhos uma posição para a direita. A chave mediana de `y` sobe para `x`, e o novo nó `z` é conectado como filho de `x`.

#### 4. Função `Insere_Nao_Completo`

Desce a árvore garantindo que nunca entraremos num nó cheio.

* **Passo 1 (Inserção na Folha):** Se `x->folha` for verdadeiro, apenas empurramos as chaves maiores para a direita e inserimos a chave `k` na posição correta, pois sabemos que a folha tem espaço.
* **Passo 2 (Descida):** Se não for folha, achamos o índice `i` do filho por onde devemos descer.
* **Passo 3 (Verificação Preemptiva):** `if(x->filho[i]->nchaves == ((2*t)-1))` – *"Divide-se cada nó completo encontrado"*. Antes de descer para o filho, checamos se ele está cheio. Se estiver, chamamos `Divisao`. Após dividir, verificamos de qual lado da nova chave mediana devemos descer (`if(k>x->chave[i]) i++;`).
* **Passo 4:** A função é chamada recursivamente no filho seguro: `Insere_Nao_Completo(x->filho[i], k, t);`.
